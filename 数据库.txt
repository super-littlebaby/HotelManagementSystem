-- ============================================================
-- 酒店管理系统 - 数据库表结构
-- 数据库：SQL Server 2016+
-- ============================================================

-- 1. 酒店信息表
CREATE TABLE hotels (
    id INT IDENTITY(1,1) PRIMARY KEY,
    name VARCHAR(200) NOT NULL,              -- 酒店名称
    address VARCHAR(300) NULL,               -- 地址
    phone VARCHAR(20) NULL,                  -- 电话
    email VARCHAR(100) NULL,                 -- 邮箱
    description VARCHAR(MAX) NULL            -- 简介
);

-- 2. 房型表（基础房价直接定义在此）
CREATE TABLE room_types (
    id INT IDENTITY(1,1) PRIMARY KEY,
    hotel_id INT NOT NULL,
    name VARCHAR(100) NOT NULL,              -- 例如：标准大床房
    description VARCHAR(MAX) NULL,
    max_adults INT DEFAULT 2,
    max_children INT DEFAULT 1,
    base_price DECIMAL(10,2) NOT NULL,       -- 每晚基础价格（不再设置每日价格计划）
    area DECIMAL(5,1) NULL,                  -- 面积(㎡)
    bed_type VARCHAR(50) NULL,               -- 床型
    CONSTRAINT fk_room_types_hotel FOREIGN KEY (hotel_id) REFERENCES hotels(id) ON DELETE CASCADE
);

-- 3. 设施表（记录客房内可损坏物品及赔偿价格）
CREATE TABLE facilities (
    id INT IDENTITY(1,1) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,              -- 设施名称，如“电视机”
    price DECIMAL(10,2) NOT NULL DEFAULT 0   -- 损坏赔偿单价
);

-- 4. 房型-设施关联表（某房型包含哪些设施）
CREATE TABLE room_type_facilities (
    room_type_id INT NOT NULL,
    facility_id INT NOT NULL,
    CONSTRAINT pk_room_type_facilities PRIMARY KEY (room_type_id, facility_id),
    CONSTRAINT fk_rtf_room_type FOREIGN KEY (room_type_id) REFERENCES room_types(id) ON DELETE CASCADE,
    CONSTRAINT fk_rtf_facility FOREIGN KEY (facility_id) REFERENCES facilities(id) ON DELETE CASCADE
);

-- 5. 房间表（具体物理房间）
CREATE TABLE rooms (
    id INT IDENTITY(1,1) PRIMARY KEY,
    hotel_id INT NOT NULL,                     -- 所属酒店ID
    room_type_id INT NOT NULL,
    room_number VARCHAR(10) NOT NULL UNIQUE,  -- 房间号
    floor INT NULL,
    status VARCHAR(20) DEFAULT 'vacant'
        CHECK (status IN ('vacant','occupied','dirty','out_of_order')),
    notes VARCHAR(MAX) NULL,
    CONSTRAINT fk_rooms_hotel FOREIGN KEY (hotel_id) REFERENCES hotels(id),
    CONSTRAINT fk_rooms_room_type FOREIGN KEY (room_type_id) REFERENCES room_types(id)
);

-- 6. 客人档案表
CREATE TABLE guests (
    id INT IDENTITY(1,1) PRIMARY KEY,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    id_type VARCHAR(20) DEFAULT 'id_card'
        CHECK (id_type IN ('id_card','passport','drivers_license','other')),
    id_number VARCHAR(200) NULL,             -- 证件号码（加密存储）
    phone VARCHAR(20) NULL,
    email VARCHAR(100) NULL,
    password VARCHAR(200) NULL,              -- 登录密码（BCrypt加密）
    nationality VARCHAR(50) NULL,
    gender VARCHAR(10) NULL
        CHECK (gender IN ('male','female','other')),
    date_of_birth DATE NULL,
    notes VARCHAR(MAX) NULL,
    created_at DATETIME DEFAULT GETDATE()
);
CREATE INDEX idx_guests_id_number ON guests(id_number);
CREATE UNIQUE INDEX idx_guests_email ON guests(email);

-- 7. 员工表（系统用户）
CREATE TABLE employees (
    id INT IDENTITY(1,1) PRIMARY KEY,
    hotel_id INT NULL,                       -- NULL表示集团或跨酒店权限
    username VARCHAR(50) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    role VARCHAR(20) DEFAULT 'front_desk'
        CHECK (role IN ('admin','manager','front_desk','housekeeping','finance')),
    first_name VARCHAR(50) NULL,
    last_name VARCHAR(50) NULL,
    phone VARCHAR(20) NULL,
    email VARCHAR(100) NULL,
    hire_date DATE NULL,
    is_active BIT DEFAULT 1,
    CONSTRAINT fk_employees_hotel FOREIGN KEY (hotel_id) REFERENCES hotels(id) ON DELETE SET NULL
);

-- 8. 预订表
CREATE TABLE reservations (
    id INT IDENTITY(1,1) PRIMARY KEY,
    guest_id INT NOT NULL,
    hotel_id INT NULL,                        -- 所属酒店ID（用于数据隔离）
    booking_date DATETIME DEFAULT GETDATE(),
    check_in_date DATE NOT NULL,
    check_out_date DATE NOT NULL,
    status VARCHAR(20) DEFAULT 'pending'
        CHECK (status IN ('pending','confirmed','checked_in','cancelled','no_show')),
    total_amount DECIMAL(10,2) NULL,          -- 预订时估算总金额
    special_requests VARCHAR(MAX) NULL,
    employee_id INT NULL,                     -- 操作员工（线下预订）
    channel VARCHAR(20) DEFAULT 'online'
        CHECK (channel IN ('online','phone','walk_in','ota')),
    CONSTRAINT fk_reservations_guest FOREIGN KEY (guest_id) REFERENCES guests(id),
    CONSTRAINT fk_reservations_hotel FOREIGN KEY (hotel_id) REFERENCES hotels(id) ON DELETE SET NULL,
    CONSTRAINT fk_reservations_employee FOREIGN KEY (employee_id) REFERENCES employees(id) ON DELETE SET NULL
);

-- 9. 预订房间明细（支持一次预订多间）
CREATE TABLE reservation_rooms (
    id INT IDENTITY(1,1) PRIMARY KEY,
    reservation_id INT NOT NULL,
    room_type_id INT NOT NULL,
    room_id INT NULL,                        -- 预分配房间，可空
    adults INT DEFAULT 1,
    children INT DEFAULT 0,
    rate_per_night DECIMAL(10,2) NOT NULL,   -- 预订时锁定的房价
    CONSTRAINT fk_resrooms_reservation FOREIGN KEY (reservation_id) REFERENCES reservations(id) ON DELETE CASCADE,
    CONSTRAINT fk_resrooms_room_type FOREIGN KEY (room_type_id) REFERENCES room_types(id),
    CONSTRAINT fk_resrooms_room FOREIGN KEY (room_id) REFERENCES rooms(id) ON DELETE SET NULL
);

-- 10. 入住登记表
CREATE TABLE check_ins (
    id INT IDENTITY(1,1) PRIMARY KEY,
    reservation_id INT NULL,                 -- 关联预订，walk-in 为空
    guest_id INT NULL,                       -- 主登记客人（线下客户可为空）
    guest_name VARCHAR(100) NULL,            -- 线下客户姓名
    id_type VARCHAR(20) NULL                 -- 证件类型：id_card/passport/drivers_license/other
        CHECK (id_type IN ('id_card','passport','drivers_license','other')),
    id_number VARCHAR(200) NULL,             -- 证件号码（加密存储）
    phone VARCHAR(20) NULL,                  -- 联系电话
    room_id INT NOT NULL,
    adults INT DEFAULT 1,
    children INT DEFAULT 0,
    check_in_time DATETIME NOT NULL,
    expected_check_out_time DATETIME NULL,
    actual_check_out_time DATETIME NULL,
    status VARCHAR(20) DEFAULT 'in_house'
        CHECK (status IN ('in_house','checked_out','early_check_out')),
    rate_per_night DECIMAL(10,2) NULL,       -- 实际执行房价
    total_charge DECIMAL(10,2) NULL,         -- 累计总费用（房费+消费+损坏）
    notes VARCHAR(MAX) NULL,
    CONSTRAINT fk_checkins_reservation FOREIGN KEY (reservation_id) REFERENCES reservations(id) ON DELETE SET NULL,
    CONSTRAINT fk_checkins_guest FOREIGN KEY (guest_id) REFERENCES guests(id) ON DELETE SET NULL,
    CONSTRAINT fk_checkins_room FOREIGN KEY (room_id) REFERENCES rooms(id)
);

-- 11. 同住客人表（记录所有在住人员）
CREATE TABLE stay_guests (
    id INT IDENTITY(1,1) PRIMARY KEY,
    check_in_id INT NOT NULL,
    guest_id INT NULL,                        -- 关联客人账号（线下客户可为空）
    name VARCHAR(100) NULL,                   -- 线下客户姓名
    id_type VARCHAR(20) NULL                  -- 证件类型：id_card/passport/drivers_license/other
        CHECK (id_type IN ('id_card','passport','drivers_license','other')),
    id_number VARCHAR(200) NULL,              -- 证件号码（加密存储）
    is_primary BIT DEFAULT 0,                 -- 是否主登记人
    CONSTRAINT fk_stayguests_checkin FOREIGN KEY (check_in_id) REFERENCES check_ins(id) ON DELETE CASCADE,
    CONSTRAINT fk_stayguests_guest FOREIGN KEY (guest_id) REFERENCES guests(id) ON DELETE SET NULL
);

-- 12. 可消费项目定义表（餐饮、洗衣等）
CREATE TABLE consumable_items (
    id INT IDENTITY(1,1) PRIMARY KEY,
    hotel_id INT NULL,
    name VARCHAR(100) NOT NULL,              -- 品名，如“自助早餐”“干洗西装”
    category VARCHAR(20) NOT NULL
        CHECK (category IN ('food','beverage','laundry','other')),
    price DECIMAL(10,2) NOT NULL,
    is_active BIT DEFAULT 1,
    CONSTRAINT fk_consumable_hotel FOREIGN KEY (hotel_id) REFERENCES hotels(id) ON DELETE CASCADE
);

-- 13. 账单表（每笔入住对应一张主账单）
CREATE TABLE bills (
    id INT IDENTITY(1,1) PRIMARY KEY,
    check_in_id INT NOT NULL,
    bill_status VARCHAR(20) DEFAULT 'open'
        CHECK (bill_status IN ('open','closed','void')),
    total_amount DECIMAL(10,2) DEFAULT 0.00,  -- 账单总金额（房费+消费+损坏）
    paid_amount DECIMAL(10,2) DEFAULT 0.00,   -- 已付金额（含押金）
    deposit_amount DECIMAL(10,2) DEFAULT 0.00,-- 押金金额（入住时收取）
    created_at DATETIME DEFAULT GETDATE(),
    closed_at DATETIME NULL,
    CONSTRAINT fk_bills_checkin FOREIGN KEY (check_in_id) REFERENCES check_ins(id)
);

-- 14. 账单明细（记录每一笔消费或损坏）
CREATE TABLE bill_items (
    id INT IDENTITY(1,1) PRIMARY KEY,
    bill_id INT NOT NULL,
    item_type VARCHAR(20) NOT NULL
        CHECK (item_type IN ('room_charge','food','beverage','laundry','damage','other')),
    description VARCHAR(255) NULL,
    facility_id INT NULL,                    -- 如果是损坏，关联损坏的设施
    consumable_id INT NULL,                  -- 如果是餐饮/洗衣，关联消费项目
    quantity DECIMAL(10,2) DEFAULT 1,
    unit_price DECIMAL(10,2) NOT NULL,
    amount DECIMAL(10,2) NOT NULL,           -- 小计
    charge_date DATE NULL,
    employee_id INT NULL,
    CONSTRAINT fk_billitems_bill FOREIGN KEY (bill_id) REFERENCES bills(id) ON DELETE CASCADE,
    CONSTRAINT fk_billitems_facility FOREIGN KEY (facility_id) REFERENCES facilities(id) ON DELETE SET NULL,
    CONSTRAINT fk_billitems_consumable FOREIGN KEY (consumable_id) REFERENCES consumable_items(id) ON DELETE SET NULL,
    CONSTRAINT fk_billitems_employee FOREIGN KEY (employee_id) REFERENCES employees(id) ON DELETE SET NULL
);

-- 15. 收款记录表（包含押金和消费付款）
CREATE TABLE payments (
    id INT IDENTITY(1,1) PRIMARY KEY,
    bill_id INT NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    payment_method VARCHAR(20) NOT NULL
        CHECK (payment_method IN ('cash','credit_card','debit_card','wechat','alipay','bank_transfer')),
    payment_type VARCHAR(20) DEFAULT 'charge'
        CHECK (payment_type IN ('deposit','charge')),  -- deposit=押金，charge=其他付款
    payment_date DATETIME DEFAULT GETDATE(),
    transaction_ref VARCHAR(100) NULL,
    employee_id INT NULL,
    CONSTRAINT fk_payments_bill FOREIGN KEY (bill_id) REFERENCES bills(id),
    CONSTRAINT fk_payments_employee FOREIGN KEY (employee_id) REFERENCES employees(id) ON DELETE SET NULL
);

-- 16. 退款记录表（退押金时使用）
CREATE TABLE refunds (
    id INT IDENTITY(1,1) PRIMARY KEY,
    bill_id INT NOT NULL,
    amount DECIMAL(10,2) NOT NULL,           -- 退款金额
    refund_method VARCHAR(20) NOT NULL
        CHECK (refund_method IN ('cash','credit_card','debit_card','wechat','alipay','bank_transfer')),
    refund_date DATETIME DEFAULT GETDATE(),
    transaction_ref VARCHAR(100) NULL,
    employee_id INT NULL,
    notes VARCHAR(255) NULL,
    CONSTRAINT fk_refunds_bill FOREIGN KEY (bill_id) REFERENCES bills(id),
    CONSTRAINT fk_refunds_employee FOREIGN KEY (employee_id) REFERENCES employees(id) ON DELETE SET NULL
);

-- 17. 房间状态变更日志（跟踪房态变化）
CREATE TABLE room_status_logs (
    id INT IDENTITY(1,1) PRIMARY KEY,
    room_id INT NOT NULL,
    old_status VARCHAR(20) NULL,
    new_status VARCHAR(20) NOT NULL,
    changed_by INT NULL,
    changed_at DATETIME DEFAULT GETDATE(),
    notes VARCHAR(MAX) NULL,
    CONSTRAINT fk_statuslog_room FOREIGN KEY (room_id) REFERENCES rooms(id) ON DELETE CASCADE,
    CONSTRAINT fk_statuslog_employee FOREIGN KEY (changed_by) REFERENCES employees(id) ON DELETE SET NULL
);
